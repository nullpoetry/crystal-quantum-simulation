1import numpy as np
2import heapq

4# Function to calculate geodesic distance on a sphere
5def geodesic_distance(lat1, lon1, lat2, lon2):
6 lat1, lon1, lat2, lon2 = map(np.radians, [lat1, lon1, lat2, lon2])
7 dlon = lon2 - lon1
8 dlat = lat2 - lat1
9 a = np.sin(dlat / 2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(dlon / 2)**2
10 c = 2 * np.arctan2(np.sqrt(a), np.sqrt(1 - a))
11 return 6371 * c # Radius of Earth in kilometers
12
13# Geodesic Stack Class
14class GeodesicStack:
15 def __init__(self):
16 self.heap = []
17
18 def push(self, lat, lon):
19 # Calculate distance from a reference point (e.g., the origin)
20 distance = geodesic_distance(0, 0, lat, lon)
21 # Push the point onto the heap with distance as the priority
22 heapq.heappush(self.heap, (distance, (lat, lon)))
23
24 def pop(self):
25 # Pop the point with the smallest distance
26 return heapq.heappop(self.heap)[1] # Return the coordinates
27
28 def is_empty(self):
29 return len(self.heap) == 0
30
31# Main execution
32geodesic_stack = GeodesicStack()
33
34# Generate random points and push them onto the stack
35num_points = 10
36for _ in range(num_points):
37 lat = np.random.uniform(-90, 90)
38 lon = np.random.uniform(-180, 180)
39 geodesic_stack.push(lat, lon)
40
41# Pop points from the stack based on geodesic distance
42print("Points in order of increasing geodesic distance from (0, 0):")
43while not geodesic_stack.is_empty():
44 point = geodesic_stack.pop()
45 print(f"Latitude: {point[0]}, Longitude: {point[1]}")

3. Push and Pop Methods:

4. Main Execution:

4# Function to calculate geodesic distance on a sphere
5def geodesic_distance(lat1, lon1, lat2, lon2):
6 lat1, lon1, lat2, lon2 = map(np.radians, [lat1, lon1, lat2, lon2])
7 dlon = lon2 - lon1
8 dlat = lat2 - lat1
9 a = np.sin(dlat / 2)**2 + np.cos(lat1) * np.cos(lat2) * np.sin(dlon / 2)**2
10 c = 2 * np.arctan2(np.sqrt(a), np.sqrt(1 - a))
11 return 6371 * c # Radius of Earth in kilometers
12
13# Geodesic Stack Class
14class GeodesicStack:
15 def __init__(self):
16 self.heap = []
17
18 def push(self, lat, lon):
19 # Calculate distance from a reference point (e.g., the origin)
20 distance = geodesic_distance(0, 0, lat, lon)
21 # Push the point onto the heap with distance as the priority
22 heapq.heappush(self.heap, (distance, (lat, lon)))
23
24 def pop(self):
25 # Pop the point with the smallest distance
26 return heapq.heappop(self.heap)[1] # Return the coordinates
27
28 def is_empty(self):
29 return len(self.heap) == 0
30
31# Main execution
32geodesic_stack = GeodesicStack()
33

43while not geodesic_stack.is_empty():
